  <script>
    function setScale(val, btn) {
      document.documentElement.style.setProperty('--p-scale', val);
      document.querySelectorAll('.sz-btn').forEach(b => b.classList.remove('active'));
      if (btn) btn.classList.add('active');
    }

    window.onload = function() {
      var p = new URLSearchParams(window.location.search);
      var rm   = (p.get('rm')  || '').trim();
      var din  = (p.get('d')   || '--:--').trim();
      var brk  = (p.get('b')   || '--:--').trim();
      var outT = (p.get('out') || '10:00').trim();
      var pin  = (p.get('pin') || '1234').trim();
      var tan  = (p.get('t')   || '赤').trim();
      var zone = (p.get('z')   || 'main').trim();

      var rmText = rm ? (rm + ' 号室') : 'ご宿泊ゲスト様';
      document.getElementById('rmDisp1').innerText = rmText;
      document.getElementById('rmDisp2').innerText = rmText;
      document.getElementById('dinDisp').innerText = din;
      document.getElementById('brkDisp').innerText = brk;
      document.getElementById('outDisp1').innerText = outT;
      document.getElementById('outDisp2').innerText = outT;
      document.getElementById('pinDisp').innerText = pin;
      document.getElementById('tanDisp').innerText = tan;

      var routeHtml = '🏢 <b>【本館 2・5・6・7階（201〜208 / 501〜714号室）】</b> 本館エレベーターにて各客室階へお進みください。';
      if (zone === 'east') {
        routeHtml = '🌉 <b>【東館 2・3階（2201〜2207 / 2301〜2307号室）】</b> <b>東館2階と本館4階（フロント）が同じ高さで連絡通路で結ばれております。</b>フロント階奥、「茶色の吊り看板」のところを左手に出て頂くと連絡通路です。';
      } else if (zone === 'main89') {
        routeHtml = '🛗 <b>【本館 8・9階（801〜808 / 901〜908号室）重要】</b> 8・9階まで上がれるエレベーターは、<b>エントランス付近の一機のみ</b>でございます。';
      }
      document.getElementById('routeDisp').innerHTML = routeHtml;

      if (p.get('print') === '1') {
        setTimeout(function() { window.print(); }, 250);
      }
    };
  </script>
